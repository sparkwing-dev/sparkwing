package store

import "testing"

func TestRetryAttemptFenceProbe(t *testing.T) {
	tenant, _, actor := liveHolderFixture(t)
	rewriteSchemaExec(t, tenant.s.db, `UPDATE nodes SET execution_json=? WHERE team='a' AND run_id='run' AND node_id='job'`, []byte(`{"retry":{"set":true,"value":2}}`))
	change := NodeTransition{Ref: NodeRef{RunID: "run", NodeID: "job"}, To: NodeRunning, Actor: actor, Cause: CauseRetry, Extra: TransitionExtra{Result: &NodeResult{Outcome: NodeFailed, Reason: "verify", MaskedError: "same failure"}}}
	for attempt := int64(4); attempt <= 5; attempt++ {
		got, err := tenant.TransitionNode(t.Context(), change)
		if err != nil || got.Replayed || got.Attempt != attempt {
			t.Fatalf("failure of attempt %d: got %+v, err=%v; want new attempt %d", attempt-1, got, err, attempt)
		}
	}
}
