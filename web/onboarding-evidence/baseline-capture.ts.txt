import { chromium } from '/var/tmp/sw-lead-tmp/onboarding-wt/web/node_modules/@playwright/test';
import { startStaticDashboard } from '/tmp/sw-codex-onboarding/baseline-server';
import { startAuthenticatedDashboard } from '/var/tmp/sw-lead-tmp/onboarding-wt/web/tests/browser/authenticated-server';
import { onboardingFixture } from '/var/tmp/sw-lead-tmp/onboarding-wt/web/tests/browser/onboarding-fixture';
import { mkdir, writeFile } from 'node:fs/promises';
import AxeBuilder from '/var/tmp/sw-lead-tmp/onboarding-wt/web/node_modules/@axe-core/playwright';
const stage = process.argv[2] || 'before';
const root = `/var/tmp/sw-lead-tmp/onboarding-wt/web/onboarding-evidence/${stage}`;
async function shot(page, options) {
 await page.screenshot(options);
 if (stage === 'after') {
  const result = await new AxeBuilder({page}).withTags(['wcag2a','wcag2aa','wcag21aa']).analyze();
  await writeFile(options.path.replace('.png','.axe.json'), JSON.stringify(result,null,2));
  if (result.violations.length) throw new Error(JSON.stringify(result.violations));
 }
}
async function main() {
 await mkdir(root, {recursive:true});
 const browser = await chromium.launch();
 const server = await startStaticDashboard();
 const auth = await startAuthenticatedDashboard({githubApp:true});
 try {
  for(const [size,width,height] of [['desktop',1440,1100],['mobile',390,844]] as const) {
   const context = await browser.newContext({viewport:{width,height}});
   const page = await context.newPage();
   await page.route('**/*', route => new URL(route.request().url()).hostname === '127.0.0.1' ? route.continue() : route.abort());
   await page.goto(`${auth.origin}/login`);
   await page.getByRole('main').waitFor();
   await shot(page,{path:`${root}/${size}-01-sign-in.png`,fullPage:true});
   await page.context().addCookies([{name:'sw_csrf',value:'fixture-csrf',url:server.origin}]);
   const fixture = await onboardingFixture(page,{connected:false});
   for (const [name,path] of [['02-create-team','/team/new'],['03-team','/team'],['04-connect-github','/team/github']]) {
    await page.goto(server.origin+path);
    await page.getByRole('heading').first().waitFor();
    await shot(page,{path:`${root}/${size}-${name}.png`,fullPage:true});
   }
   fixture.connect();
   await page.reload();
   await page.getByRole('combobox',{name:'Repository to set up',exact:true}).selectOption('acme/app');
   await page.getByText('release-frontend-ci',{exact:true}).waitFor();
   await shot(page,{path:`${root}/${size}-05-repository.png`,fullPage:true});
   if(stage==='before') {
    await page.getByRole('combobox',{name:'Runs on',exact:true}).selectOption('github-actions');
    await page.getByRole('button',{name:'Enable repository declarations',exact:true}).click();
    await page.getByRole('button',{name:'Allow GitHub Actions execution',exact:true}).click();
    await page.getByText('View generated workflow',{exact:true}).click();
   } else {
    await page.getByRole('button',{name:'Enable runs on GitHub Actions',exact:true}).click();
    await page.getByText('Workflow and team variable',{exact:true}).waitFor();
    await page.getByText('Workflow and team variable',{exact:true}).scrollIntoViewIfNeeded();
   }
   await shot(page,{path:`${root}/${size}-06-workflow.png`,fullPage:true});
   if(stage==='after') {
    await page.getByLabel('My enabled workflow starts a Sparkwing runner on pushes for team korey-gam',{exact:true}).check();
    await page.getByRole('heading',{name:'Ready for your next push'}).scrollIntoViewIfNeeded();
    await shot(page,{path:`${root}/${size}-07-ready.png`,fullPage:true});
   }
   await page.goto(server.origin+'/team/machines');
   await page.getByRole('heading',{name:'CLI access',exact:true}).scrollIntoViewIfNeeded();
   await shot(page,{path:`${root}/${size}-08-cli-machines.png`,fullPage:true});
   await page.goto(server.origin+'/device');
   await page.getByRole('heading',{name:'Sign in a terminal'}).waitFor();
   await shot(page,{path:`${root}/${size}-09-device.png`,fullPage:true});
   await page.getByLabel('Device code',{exact:true}).fill('BCDF-GHJK');
   await page.getByRole('button',{name:'Continue',exact:true}).click();
   await page.getByRole('heading',{name: "Approve sign-in for Korey's desktop?",exact:true}).waitFor();
   await shot(page,{path:`${root}/${size}-10-device-approval.png`,fullPage:true});
   await context.close();
  }
 } finally { await auth.close(); await server.close(); await browser.close(); }
}
main().catch(e=>{console.error(e);process.exitCode=1});
