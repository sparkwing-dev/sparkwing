type Config struct {
	Defaults  *Defaults
	Pipelines []Pipeline
	Sparks    []Spark
}
func (c *Config) Registry() *pipelines.Config
type Defaults struct {
	Run      *RunRestrictions
	Args     map[string]string
	Guards   *Guards
	Requires []string
	Prefers  []string
}
func Discover(startDir string) (string, *Config, error)
type Events struct {
	Push        *PushEvent
	PullRequest *PullRequestEvent
	Schedule    []Schedule
	PreCommit   *Hook
	PrePush     *Hook
	PostCommit  *Hook
}

	When   []string

}

type Pipeline struct {

	Name        string
	Entrypoint  string
	Description *string
	Hidden      *bool
	Source      *string
	Profile     *string
	Args        map[string]string
	Guards      *Guards
	Requires    []string
	Prefers     []string
	SourceFiles []string

}
func PipelineSource(repoDir, name string) (string, error)

	Actions  []string

}
type PushEvent struct {

}

}

	Name    *string

}

}
